cmake_minimum_required(VERSION 3.22)
project(DrumFilter VERSION 0.3.1 LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 17)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
include(FetchContent)
FetchContent_Declare(JUCE GIT_REPOSITORY https://github.com/juce-framework/JUCE.git GIT_TAG 8.0.10 GIT_SHALLOW TRUE)
FetchContent_MakeAvailable(JUCE)
juce_add_plugin(DrumFilter COMPANY_NAME LM18 PLUGIN_MANUFACTURER_CODE Lm18 PLUGIN_CODE DrFl FORMATS VST3 PRODUCT_NAME "DrumFilter" IS_SYNTH FALSE NEEDS_MIDI_INPUT FALSE NEEDS_MIDI_OUTPUT FALSE COPY_PLUGIN_AFTER_BUILD FALSE)
target_sources(DrumFilter PRIVATE Source/PluginProcessor.cpp Source/PluginProcessor.h)
target_compile_definitions(DrumFilter PRIVATE JUCE_WEB_BROWSER=0 JUCE_USE_CURL=0)
target_link_libraries(DrumFilter PRIVATE juce::juce_audio_utils juce::juce_dsp juce::juce_recommended_config_flags juce::juce_recommended_warning_flags)
